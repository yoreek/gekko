import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid stagger>
  <Card title="Kein erneutes Flashen pro Projekt" icon="setting">
    Füge Schalter, Sensoren, Displays, Zeitpläne und Pumpen zur Laufzeit über
    das Webportal hinzu. Geräteregistrierung, Abhängigkeitsgraph und
    Dashboard-Layout befinden sich vollständig auf dem Gerät — niemals in
    einer YAML-Datei, die neu kompiliert werden muss.
  </Card>
  <Card title="Local-first" icon="laptop">
    Das Portal wird aus dem Flash-Speicher des ESP32 über WiFi bereitgestellt.
    MQTT, Home-Assistant-Discovery und OTA-Updates sind optional und
    standardmäßig deaktiviert.
  </Card>
  <Card title="Typisierte Geräteregistrierung" icon="puzzle">
    Geräte deklarieren Abhängigkeiten untereinander — ein Schalter hinter
    einem Portexpander, ein Sensor hinter einem I2C-Bus oder eine von einem
    Zeitplan gesteuerte Pumpe. Die Registrierung prüft und speichert diesen
    Graphen.
  </Card>
  <Card title="Displays, Zeitpläne und Dosierung" icon="rocket">
    Pixel-, Zeichen- und Siebensegmentanzeigen mit visuellem Layout-Designer,
    minutengenaue Zeitpläne, bedingungsgesteuerte Automatisierung, Thermostate
    sowie Dosierpumpen mit Kalibrierung und Verlauf.
  </Card>
  <Card title="Home Assistant mit einem Schalter" icon="heart">
    Aktiviere einen Schalter auf der Geräteseite und es erscheint in Home
    Assistant als native Entität — steuerbar aus HA und weiterhin lokal
    ausgeführt.
  </Card>
  <Card title="Sicherung und Wiederherstellung" icon="document">
    Exportiere die vollständige Gerätekonfiguration als ein einziges,
    menschenlesbares Paket und stelle sie auf jedem Gekko-Controller wieder
    her.
  </Card>
  <Card title="Spricht deine Sprache" icon="translate">
    Das Portal ist auf Englisch, Ukrainisch, Russisch, Deutsch, Spanisch,
    Französisch und Italienisch verfügbar und erkennt die Browsersprache
    automatisch.
  </Card>
</CardGrid>
